   383→/* ================= Layar Kasir ================= */
   384→export function KasirView() {
   385→  const products = useZPos((s) => s.products)
   386→  const categories = useZPos((s) => s.categories)
   387→  const cart = useZPos((s) => s.cart)
   388→  const settings = useZPos((s) => s.settings)
   389→  const addToCart = useZPos((s) => s.addToCart)
   390→  const orderDiscount = useZPos((s) => s.orderDiscount)
   391→
   392→  const [search, setSearch] = useState('')
   393→  const [activeCat, setActiveCat] = useState<string>('all')
   394→  const [paymentOpen, setPaymentOpen] = useState(false)
   395→  const [sheetOpen, setSheetOpen] = useState(false)
   396→  const [receipt, setReceipt] = useState<Sale | null>(null)
   397→  const searchRef = useRef<HTMLInputElement>(null)
   398→
   399→  const totals = computeTotals(cart, orderDiscount, settings)
   400→  const itemCount = cart.reduce((s, c) => s + c.qty, 0)
   401→
   402→  // Shortcut: "/" fokus cari · F9 bayar · Esc tutup sheet
   403→  useEffect(() => {
   404→    const handler = (e: KeyboardEvent) => {
   405→      const tag = document.activeElement?.tagName
   406→      if (e.key === '/' && tag !== 'INPUT' && tag !== 'TEXTAREA') {
   407→        e.preventDefault()
   408→        searchRef.current?.focus()
   409→      }
   410→      if (e.key === 'F9') {
   411→        e.preventDefault()
   412→        if (useZPos.getState().cart.length > 0) setPaymentOpen(true)
   413→      }
   414→    }
   415→    window.addEventListener('keydown', handler)
   416→    return () => window.removeEventListener('keydown', handler)
   417→  }, [])
   418→
   419→  const catCounts = useMemo(() => {
   420→    const m = new Map<string, number>()
   421→    for (const p of products) {
   422→      if (!p.isActive) continue
   423→      const key = p.categoryId ?? 'none'
   424→      m.set(key, (m.get(key) ?? 0) + 1)
   425→    }
   426→    return m
   427→  }, [products])
   428→
   429→  const filtered = useMemo(() => {
   430→    const q = search.trim().toLowerCase()
   431→    return products.filter((p) => {
   432→      if (!p.isActive) return false
   433→      if (activeCat !== 'all' && p.categoryId !== activeCat) return false
   434→      if (!q) return true
   435→      return (
   436→        p.name.toLowerCase().includes(q) ||
   437→        p.sku.toLowerCase().includes(q) ||
   438→        (p.barcode ?? '').includes(q)
   439→      )
   440→    })
   441→  }, [products, search, activeCat])
   442→
   443→  const onSearchEnter = () => {
   444→    const q = search.trim().toLowerCase()
   445→    if (!q) return
   446→    const exact = products.find(
   447→      (p) => p.isActive && ((p.barcode ?? '').toLowerCase() === q || p.sku.toLowerCase() === q)
   448→    )
   449→    if (exact) {
   450→      addToCart(exact)
   451→      setSearch('')
   452→    }
   453→  }
   454→
   455→  const openPayment = () => {
   456→    setSheetOpen(false)
   457→    setPaymentOpen(true)
   458→  }
   459→
   460→  const catChips = [
   461→    { id: 'all', name: 'Semua', count: products.filter((p) => p.isActive).length },
   462→    ...categories.map((c) => ({ id: c.id, name: c.name, count: catCounts.get(c.id) ?? 0 })),
   463→  ]
   464→
   465→  return (
   466→    <div className="flex h-full min-h-0">
   467→      {/* ============ Katalog ============ */}
   468→      <section className="flex min-w-0 flex-1 flex-col" aria-label="Katalog produk">
   469→        <div className="shrink-0 border-b border-zinc-200 bg-white px-4 py-3">
   470→          <div className="flex items-center gap-2.5">
   471→            <div className="relative flex-1">
   472→              <ScanBarcode className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-zinc-400" aria-hidden="true" />
   473→              <Input
   474→                ref={searchRef}
   475→                value={search}
   476→                onChange={(e) => setSearch(e.target.value)}
   477→                onKeyDown={(e) => e.key === 'Enter' && onSearchEnter()}
   478→                placeholder="Cari nama / SKU / scan barcode…"
   479→                className="h-10 border-zinc-200 bg-zinc-50 pl-9 pr-14 text-sm focus-visible:bg-white"
   480→                aria-label="Cari produk"
   481→              />
   482→              {search ? (
   483→                <button
   484→                  onClick={() => setSearch('')}
   485→                  className="absolute right-2.5 top-1/2 -translate-y-1/2 text-zinc-400 hover:text-zinc-700"
   486→                  aria-label="Hapus pencarian"
   487→                >
   488→                  <X className="h-4 w-4" />
   489→                </button>
   490→              ) : (
   491→                <span className="absolute right-2.5 top-1/2 hidden -translate-y-1/2 sm:block"><Kbd>/</Kbd></span>
   492→              )}
   493→            </div>
   494→            <span className="hidden shrink-0 rounded-lg border border-zinc-200 bg-white px-2.5 py-1.5 text-xs font-semibold text-zinc-500 sm:block">
   495→              {filtered.length} produk
   496→            </span>
   497→          </div>
   498→
   499→          <div className="zpos-scroll -mx-4 mt-2.5 flex gap-1.5 overflow-x-auto px-4 pb-0.5">
   500→            {catChips.map((c) => (
   501→              <button
   502→                key={c.id}
   503→                onClick={() => setActiveCat(c.id)}
   504→                className={cn(
   505→                  'flex shrink-0 items-center gap-1.5 rounded-full border px-3.5 py-1.5 text-xs font-semibold transition-colors',
   506→                  activeCat === c.id
   507→                    ? 'border-zinc-900 bg-zinc-900 text-white'
   508→                    : 'border-zinc-200 bg-white text-zinc-600 hover:border-zinc-400'
   509→                )}
   510→              >
   511→                {c.name}
   512→                <span
   513→                  className={cn(
   514→                    'rounded-full px-1.5 py-px text-[10px] font-bold',
   515→                    activeCat === c.id ? 'bg-white/20 text-white' : 'bg-zinc-100 text-zinc-400'
   516→                  )}
   517→                >
   518→                  {c.count}
   519→                </span>
   520→              </button>
   521→            ))}
   522→          </div>
   523→        </div>
   524→
   525→        {/* Grid produk */}
   526→        <div className="zpos-scroll min-h-0 flex-1 overflow-y-auto p-4 pb-24 xl:pb-4">
   527→          <div className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-4 2xl:grid-cols-5">
   528→            {filtered.map((p) => {
   529→              const habis = p.stock <= 0
   530→              const kritis = !habis && p.stock <= p.minStock
   531→              return (
   532→                <button
   533→                  key={p.id}
   534→                  onClick={() => addToCart(p)}
   535→                  disabled={habis}
   536→                  className={cn(
   537→                    'group relative flex flex-col overflow-hidden rounded-xl border bg-white text-left shadow-sm transition-all',
   538→                    habis
   539→                      ? 'cursor-not-allowed border-zinc-200 opacity-55'
   540→                      : 'border-zinc-200 hover:-translate-y-0.5 hover:border-zinc-900 hover:shadow-lg active:scale-[0.98]'
   541→                  )}
   542→                >
   543→                  <div className="relative aspect-square w-full overflow-hidden bg-zinc-100">
   544→                    <ProductThumb
   545→                      src={p.image}
   546→                      name={p.name}
   547→                      className="h-full w-full transition-transform duration-300 group-hover:scale-105"
   548→                    />
   549→                    {habis && (
   550→                      <div className="absolute inset-0 flex items-center justify-center bg-white/70 backdrop-blur-[1px]">
   551→                        <span className="rounded-full bg-rose-600 px-3 py-1 text-[11px] font-bold text-white">HABIS</span>
   552→                      </div>
   553→                    )}
   554→                    {kritis && (
   555→                      <span className="absolute left-2 top-2 rounded-full bg-amber-500 px-2 py-0.5 text-[10px] font-bold text-white shadow-sm">
   556→                        Sisa {p.stock}
   557→                      </span>
   558→                    )}
   559→                  </div>
   560→                  <div className="flex flex-1 flex-col p-2.5">
   561→                    <p className="min-h-8 line-clamp-2 text-[13px] font-semibold leading-snug text-zinc-900">{p.name}</p>
   562→                    <div className="mt-auto flex items-baseline justify-between pt-1.5">
   563→                      <p className="text-sm font-bold tabular-nums text-zinc-900">{rupiah(p.price)}</p>
   564→                      <span className="text-[10px] font-medium text-zinc-400">{p.category?.name ?? '—'}</span>
   565→                    </div>
   566→                  </div>
   567→                </button>
   568→              )
   569→            })}
   570→            {filtered.length === 0 && (
   571→              <div className="col-span-full flex flex-col items-center gap-2 rounded-xl border border-dashed border-zinc-300 bg-white py-14 text-center">
   572→                <Search className="h-6 w-6 text-zinc-300" aria-hidden="true" />
   573→                <p className="text-sm font-semibold text-zinc-600">Tidak ada produk cocok</p>
   574→                <p className="text-xs text-zinc-400">Coba kata kunci lain atau ganti kategori.</p>
   575→              </div>
   576→            )}
   577→          </div>
   578→        </div>
   579→
   580→        {/* Bar keranjang mobile */}
   581→        <div className="shrink-0 border-t border-zinc-200 bg-white px-4 py-3 xl:hidden">
   582→          <button
   583→            onClick={() => setSheetOpen(true)}
   584→            disabled={cart.length === 0}
   585→            className="flex w-full items-center justify-between rounded-xl bg-zinc-900 px-4 py-3 text-white transition-colors hover:bg-zinc-800 disabled:opacity-40"
   586→          >
   587→            <span className="flex items-center gap-2.5">
   588→              <span className="flex h-8 w-8 items-center justify-center rounded-lg bg-white/15">
   589→                <ShoppingBag className="h-4 w-4" aria-hidden="true" />
   590→              </span>
   591→              <span className="text-left">
   592→                <span className="block text-[11px] font-medium text-zinc-300">{itemCount} item dipesan</span>
   593→                <span className="block text-sm font-bold tabular-nums">{rupiah(totals.total)}</span>
   594→              </span>
   595→            </span>
   596→            <span className="rounded-lg bg-emerald-500 px-3.5 py-2 text-sm font-bold">Lihat Keranjang</span>
   597→          </button>
   598→        </div>
   599→      </section>
   600→
   601→      {/* ============ Keranjang desktop ============ */}
   602→      <aside
   603→        className="hidden w-[380px] shrink-0 border-l border-zinc-200 bg-white 2xl:w-[400px] xl:flex xl:flex-col"
   604→        aria-label="Keranjang belanja"
   605→      >
   606→        <CartPanel onPay={openPayment} />
   607→      </aside>
   608→
   609→      {/* Keranjang mobile (sheet) */}
   610→      <Sheet open={sheetOpen} onOpenChange={setSheetOpen}>
   611→        <SheetContent side="right" className="w-full max-w-[400px] p-0 sm:max-w-[400px]">
   612→          <SheetHeader className="sr-only">
   613→            <SheetTitle>Keranjang</SheetTitle>
   614→          </SheetHeader>
   615→          <CartPanel onPay={openPayment} />
   616→        </SheetContent>
   617→      </Sheet>
   618→
   619→      <PaymentDialog
   620→        key={paymentOpen ? 'open' : 'closed'}
   621→        open={paymentOpen}
   622→        onOpenChange={setPaymentOpen}
   623→        onPaid={(sale) => {
   624→          setPaymentOpen(false)
   625→          setReceipt(sale)
   626→        }}
   627→      />
   628→      <ReceiptDialog
   629→        sale={receipt}
   630→        open={receipt !== null}
   631→        onOpenChange={(o) => !o && setReceipt(null)}
   632→      />
   633→    </div>
   634→  )
